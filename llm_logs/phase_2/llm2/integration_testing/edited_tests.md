package phase_2.llm2.edited;

import java.util.*;

public class BookScan {

    public int howManyTimes(String text, String substring) {
        if (text == null || substring == null || substring.isEmpty()) {
            return 0;
        }
        int count = 0;
        int idx = 0;
        while ((idx = text.indexOf(substring, idx)) != -1) {
            count++;
            idx++;
        }
        return count;
    }

    public int strlen(String string) {
        if (string == null) {
            return 0;
        }
        return string.length();
    }

    public String flipCase(String string) {
        if (string == null) {
            return null;
        }
        StringBuilder sb = new StringBuilder(string.length());
        for (char c : string.toCharArray()) {
            if (Character.isLowerCase(c)) {
                sb.append(Character.toUpperCase(c));
            } else if (Character.isUpperCase(c)) {
                sb.append(Character.toLowerCase(c));
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }

    public Map<String, Map<String, Object>> scanByWordLength(String text, int targetLength) {
        Map<String, Map<String, Object>> results = new LinkedHashMap<>();
        if (text == null || text.isEmpty() || targetLength <= 0) {
            return results;
        }
        String[] lines = text.split("\n");
        for (int lineIdx = 0; lineIdx < lines.length; lineIdx++) {
            String line = lines[lineIdx];
            int lineNumber = lineIdx + 1;
            String normalizedLine = toLowerUsingFlipCase(line);
            String[] tokens = line.split("\\s+");
            Set<String> processedOnLine = new HashSet<>();
            for (String token : tokens) {
                if (token.isEmpty()) {
                    continue;
                }
                String cleanWord = token.replaceAll("[^a-zA-Z0-9]", "");
                if (strlen(cleanWord) != targetLength) {
                    continue;
                }
                String key = toLowerUsingFlipCase(cleanWord);
                if (processedOnLine.contains(key)) {
                    continue;
                }
                processedOnLine.add(key);
                int occurrences = howManyTimes(normalizedLine, key);
                Map<String, Object> entry = results.computeIfAbsent(key, k -> {
                    Map<String, Object> m = new LinkedHashMap<>();
                    m.put("count", 0);
                    m.put("lines", new TreeSet<Integer>());
                    return m;
                });
                entry.put("count", (int) entry.get("count") + occurrences);
                Set<Integer> lineSet = (Set<Integer>) entry.get("lines");
                lineSet.add(lineNumber);
            }
        }
        return results;
    }

    private String toLowerUsingFlipCase(String s) {
        if (s == null) {
            return null;
        }
        StringBuilder sb = new StringBuilder(s.length());
        for (char c : s.toCharArray()) {
            if (Character.isUpperCase(c)) {
                sb.append(flipCase(String.valueOf(c)));
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }
}
