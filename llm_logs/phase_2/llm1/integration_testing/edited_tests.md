package phase_2.llm1.edited;

import java.util.*;

public class BookScan {
    public static Map<String, Map<String, Object>> scanByWordLength(String text, int targetLength) {
        Map<String, Map<String, Object>> results = new HashMap<>();
        if (text == null || text.trim().isEmpty() || targetLength <= 0) {
            return results;
        }
        String[] lines = text.split("\n");
        for (int i = 0; i < lines.length; i++) {
            int lineNumber = i + 1;
            String line = lines[i];
            String normalizedLine = normalizeToLowerCase(line);
            String[] words = line.split("\\s+");
            Set<String> processedWordsInLine = new HashSet<>();
            for (String rawWord : words) {
                String cleanWord = rawWord.replaceAll("[^a-zA-Z0-9]", "");
                if (strlen(cleanWord) == targetLength) {
                    String normalizedWord = normalizeToLowerCase(cleanWord);
                    if (!processedWordsInLine.contains(normalizedWord)) {
                        processedWordsInLine.add(normalizedWord);
                        int occurrencesInLine = howManyTimes(normalizedLine, normalizedWord);
                        if (occurrencesInLine > 0) {
                            results.putIfAbsent(normalizedWord, new HashMap<>());
                            Map<String, Object> stats = results.get(normalizedWord);
                            stats.putIfAbsent("count", 0);
                            stats.putIfAbsent("lines", new TreeSet<Integer>());
                            int currentCount = (int) stats.get("count");
                            stats.put("count", currentCount + occurrencesInLine);
                            Set<Integer> lineSet = (Set<Integer>) stats.get("lines");
                            lineSet.add(lineNumber);
                        }
                    }
                }
            }
        }
        return results;
    }

    public static int howManyTimes(String text, String substring) {
        if (text == null || substring == null || substring.isEmpty()) {
            return 0;
        }
        int count = 0;
        int idx = 0;
        while ((idx = text.indexOf(substring, idx)) != -1) {
            count++;
            idx++;
        }
        return count;
    }

    public static int strlen(String string) {
        if (string == null) {
            return 0;
        }
        return string.length();
    }

    public static String flipCase(String string) {
        if (string == null) {
            return null;
        }
        StringBuilder result = new StringBuilder(string.length());
        for (char c : string.toCharArray()) {
            if (Character.isLowerCase(c)) {
                result.append(Character.toUpperCase(c));
            } else if (Character.isUpperCase(c)) {
                result.append(Character.toLowerCase(c));
            } else {
                result.append(c);
            }
        }
        return result.toString();
    }

    private static String normalizeToLowerCase(String text) {
        if (text == null) {
            return null;
        }
        StringBuilder result = new StringBuilder(text.length());
        for (char c : text.toCharArray()) {
            if (Character.isUpperCase(c)) {
                result.append(flipCase(String.valueOf(c)));
            } else {
                result.append(c);
            }
        }
        return result.toString();
    }
}
